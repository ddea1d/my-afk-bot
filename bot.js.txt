const mineflayer = require('mineflayer');

// Функция, которая создает бота и настраивает его поведение
function createMyBot() {
    const bot = mineflayer.createBot({
        host: 'NikitaKsmp.aternos.me', // Пример: 'mycoolserver.aternos.me'
        port: 25565,                    // Порт (оставьте 25565, если у вас стандартный)
        username: 'SuperAFK_Bot',       // Ник, под которым бот зайдет на сервер
        version: '1.21.11'               // Версия сервера (можно убрать, тогда бот выберет сам)
    });

    // Действие при успешном входе на сервер
    bot.on('spawn', () => {
        console.log('--- БОТ УСПЕШНО ЗАШЕЛ НА СЕРВЕР! ---');
        
        // Если на сервере нужна регистрация/авторизация, уберіть две косые черты ниже:
        // bot.chat('/login ТВОЙ_ПАРОЛЬ_ОТ_БОТА');
    });

    // Периодическое движение, чтобы сервер не кикнул за АФК
    bot.on('time', () => {
        // Раз в несколько минут бот будет прыгать
        if (bot.time.age % 400 === 0) {
            bot.setControlState('jump', true);
            setTimeout(() => bot.setControlState('jump', false), 500);
        }
    });

    // Обработка ошибок (чтобы скрипт не вылетал, если сервер выключен)
    bot.on('error', (err) => {
        console.log('Произошла ошибка:', err.message);
    });

    // Авто-переподключение: если бота кикнули, он зайдет снова через 10 секунд
    bot.on('end', () => {
        console.log('Бот отключился от сервера. Переподключение через 10 секунд...');
        setTimeout(() => {
            createMyBot();
        }, 10000);
    });
}

// Запуск бота
createMyBot();
